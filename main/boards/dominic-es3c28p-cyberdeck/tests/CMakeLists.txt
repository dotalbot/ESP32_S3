cmake_minimum_required(VERSION 3.20)
project(cyberdeck_touch_test C CXX)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(LVGL "${REPO_ROOT}/managed_components/lvgl__lvgl")
file(GLOB_RECURSE LVGL_SOURCES "${LVGL}/src/*.c")
add_library(lvgl STATIC ${LVGL_SOURCES})
target_include_directories(lvgl PUBLIC "${LVGL}" "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(lvgl PUBLIC LV_CONF_INCLUDE_SIMPLE LV_KCONFIG_IGNORE)
add_executable(touch_rotation_test touch_rotation_test.cc)
target_compile_features(touch_rotation_test PRIVATE cxx_std_17)
target_link_libraries(touch_rotation_test PRIVATE lvgl)
file(STRINGS "${REPO_ROOT}/main/boards/dominic-fnk0104n-cyberdeck/config.h" PANEL_MIRRORS
     REGEX "^#define DISPLAY_MIRROR_[XY] ")
foreach(line IN LISTS PANEL_MIRRORS)
    string(REGEX REPLACE "^#define ([^ ]+) (.*)$" "\\1=\\2" definition "${line}")
    target_compile_definitions(touch_rotation_test PRIVATE "${definition}")
endforeach()
enable_testing()
add_test(NAME touch_rotation COMMAND touch_rotation_test)
add_executable(display_flush_test display_flush_test.cc
    "${REPO_ROOT}/managed_components/78__xiaozhi-fonts/src/font_noto_sans_basic_16_4.c")
target_compile_features(display_flush_test PRIVATE cxx_std_17)
target_link_libraries(display_flush_test PRIVATE lvgl)
add_test(NAME display_flush COMMAND display_flush_test)
add_executable(weather_test weather_test.cc ../cyberdeck_clock.cc
    "${REPO_ROOT}/managed_components/espressif__cjson/cJSON/cJSON.c")
target_compile_features(weather_test PRIVATE cxx_std_17)
target_include_directories(weather_test PRIVATE
    "${REPO_ROOT}/managed_components/espressif__cjson/cJSON")
add_test(NAME weather COMMAND weather_test)
add_executable(clock_test clock_test.cc ../cyberdeck_clock.cc)
target_compile_features(clock_test PRIVATE cxx_std_17)
add_test(NAME clock COMMAND clock_test)
